    <script>

        // 원소 데이터 정의 (반감기를 '일(days)' 단위 기준값으로 관리)

        const isotopeData = {

            "Pu-239": { name: "플루토늄-239", halfLifeText: "24,000년", halfLifeYears: 24000, isShort: false },

            "U-238":  { name: "우라늄-238", halfLifeText: "45억년", halfLifeYears: 4500000000, isShort: false },

            "Ra-226": { name: "라듐-226", halfLifeText: "1,600년", halfLifeYears: 1600, isShort: false },

            "Rn-222": { name: "라돈-222", halfLifeText: "4일", halfLifeYears: 4 / 365.25, isShort: true, days: 4 },

            "Po-210": { name: "폴로늄-210", halfLifeText: "138일", halfLifeYears: 138 / 365.25, isShort: true, days: 138 },

            "Th-232": { name: "토륨-232", halfLifeText: "140억년", halfLifeYears: 14000000000, isShort: false },

            "Cs-137": { name: "세슘-137", halfLifeText: "30년", halfLifeYears: 30, isShort: false }

        };

        // DOM 요소

        const massSlider = document.getElementById('mass-slider');

        const massDisplay = document.getElementById('mass-display');

        const launchBtn = document.getElementById('launch-btn');

        const bomb = document.getElementById('bomb');

        const mushroomCloud = document.getElementById('mushroom-cloud');

        const cloudHead = mushroomCloud.querySelector('.head');

        const cloudStem = mushroomCloud.querySelector('.stem');

        const infoPanel = document.getElementById('info-panel');

        const infoText = document.getElementById('info-text');

        let decayChart = null;

        let animationInterval = null;

        // 슬라이더 이벤트

        massSlider.addEventListener('input', (e) => {

            massDisplay.textContent = Number(e.target.value).toLocaleString();

        });

        // 투하 버튼 이벤트

        launchBtn.addEventListener('click', () => {

            launchBtn.disabled = true;

            resetSimulation();

            // 1. 폭탄 낙하 애니메이션 (5초)

            bomb.style.display = 'block';

            bomb.style.top = '-50px';

           

            const dropDuration = 5000; // 5초

            const startTime = performance.now();

            const startTop = -50;

            const endTop = window.innerHeight - 70; // 지면 부근

            function animateDrop(currentTime) {

                const elapsed = currentTime - startTime;

                const progress = Math.min(elapsed / dropDuration, 1);

               

                // 가속도 적용 (자유낙하 느낌)

                const currentTop = startTop + (endTop - startTop) * Math.pow(progress, 2);

                bomb.style.top = `${currentTop}px`;

                if (progress < 1) {

                    requestAnimationFrame(animateDrop);

                } else {

                    bomb.style.display = 'none';

                    explode(); // 폭발 시작

                }

            }

            requestAnimationFrame(animateDrop);

        });

        function resetSimulation() {

            if (animationInterval) clearInterval(animationInterval);

            mushroomCloud.style.opacity = 0;

            infoPanel.style.display = 'none';

            if (decayChart) decayChart.destroy();

        }

        // 폭발 및 붕괴 시뮬레이션

        function explode() {

            const selectedKey = document.getElementById('isotope-select').value;

            const isotope = isotopeData[selectedKey];

            const N0 = parseFloat(massSlider.value); // 초기 질량

            // 질량에 따른 버섯구름 크기 설정 (100g ~ 10000g -> 크기 스케일링)

            const scale = 0.5 + (N0 / 10000) * 1.0; // 0.5 ~ 1.5 배 스케일

            const headSize = 120 * scale;

            const stemWidth = 30 * scale;

            const stemHeight = 150 * scale;

            cloudHead.style.width = `${headSize}px`;

            cloudHead.style.height = `${headSize * 0.7}px`;

            cloudStem.style.width = `${stemWidth}px`;

            cloudStem.style.height = `${stemHeight}px`;

            mushroomCloud.style.opacity = 1;

            infoPanel.style.display = 'block';

            // 그래프 데이터 및 시뮬레이션 시간 축 설정

            const isShort = isotope.isShort;

            const stepUnitText = isShort ? "10일" : "10년";

            const chartStepText = isShort ? "1년" : "10년";

           

            // 그래프 x축 라벨 및 y축 데이터 생성

            const labels = [];

            const dataPoints = [];

            const steps = 10; // 그래프에 표기할 10개 구간

            // 반감기 공식: N(t) = N0 * (1/2)^(t / t_half)

            const halfLife = isShort ? isotope.days : isotope.halfLifeYears;

            for (let i = 0; i <= steps; i++) {

                // 반감기가 1년 이내인 원소는 1칸당 1년(365.25일), 그 외는 1칸당 10년

                let t = isShort ? i * 365.25 : i * 10;

                let N_t = N0 * Math.pow(0.5, t / halfLife);

               

                labels.push(isShort ? `${i}년` : `${i * 10}년`);

                dataPoints.push(N_t.toFixed(2));

            }

            // 차트 그려주기

            renderChart(labels, dataPoints, isotope.name);

            // 시간 경과에 따른 옅어짐 애니메이션

            // 조건: 10년당 1초 (단, 반감기가 짧은 원소는 10일당 1초)

            let elapsedTime = 0; // 초 단위 진행시간

           

            // 화면 텍스트 업데이트 함수

            function updateInfo(currentMass, passedTimeText) {

                infoText.innerHTML = `

                    <p><b>원소명:</b> ${isotope.name}</p>

                    <p><b>초기 질량 (N₀):</b> ${N0.toLocaleString()} g</p>

                    <p><b>반감기 (t<sub>1/2</sub>):</b> ${isotope.halfLifeText}</p>

                    <p><b>경과 시간:</b> ${passedTimeText}</p>

                    <p><b>남은 질량 (N):</b> <span style="color:#ff3d00; font-weight:bold;">${currentMass.toFixed(2)} g</span></p>

                `;

            }

            animationInterval = setInterval(() => {

                elapsedTime += 0.1; // 0.1초마다 계산

               

                // 1초당 흐르는 실제 시뮬레이션 상의 시간(t)

                // 반감기 짧음: 1초 = 10일 -> t_sim = elapsedTime * 10 (일)

                // 반감기 긴 원소: 1초 = 10년 -> t_sim = elapsedTime * 10 (년)

                let t_sim = elapsedTime * 10;

                let currentN = N0 * Math.pow(0.5, t_sim / halfLife);

                // 투명도 계산 (남은 질량 비율에 따라 옅어짐)

                let opacity = Math.max(0.1, currentN / N0);

                mushroomCloud.style.opacity = opacity;

                // 텍스트 시간 단위 변환

                let passedTimeText = isShort

                    ? `${(t_sim).toFixed(1)} 일`

                    : `${(t_sim).toFixed(1)} 년`;

                updateInfo(currentN, passedTimeText);

                // 질량이 거의 남아있지 않거나 30초 진행 시 멈춤

                if (currentN < 0.01 || elapsedTime > 30) {

                    clearInterval(animationInterval);

                    launchBtn.disabled = false;

                }

            }, 100);

        }

        // Chart.js를 사용한 반감기 그래프 렌더링

        function renderChart(labels, data, title) {

            const ctx = document.getElementById('decayChart').getContext('2d');

           

            decayChart = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: labels,

                    datasets: [{

                        label: `${title} 붕괴 곡선 (g)`,

                        data: data,

                        borderColor: '#ff9800',

                        backgroundColor: 'rgba(255, 152, 0, 0.2)',

                        fill: true,

                        tension: 0.3

                    }]

                },

                options: {

                    responsive: true,

                    maintainAspectRatio: false,

                    scales: {

                        x: {

                            ticks: { color: '#ccc', font: { size: 10 } },

                            grid: { color: '#444' }

                        },

                        y: {

                            ticks: { color: '#ccc', font: { size: 10 } },

                            grid: { color: '#444' },

                            title: { display: true, text: '질량 (g)', color: '#ccc' }

                        }

                    },

                    plugins: {

                        legend: { labels: { color: '#fff', font: { size: 11 } } }

                    }

                }

            });

        }

    </script>
